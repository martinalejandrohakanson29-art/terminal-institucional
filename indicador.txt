            box.delete(supportBox)

            array.remove(supportPrices, i)
            array.remove(supportBoxes, i)

// =====================================================
// FUNCIÓN ENVOLVENTE ALCISTA
// =====================================================

bullEngulfingFunc() =>
    prevBearish = close[1] < open[1]

    currentBullish = close > open

    body = math.abs(close - open)

    candleRange = high - low

    bodyPerc = candleRange > 0 ? body / candleRange * 100 : 0.0

    engulf = open <= close[1] and close >= open[1]

    prevBearish and currentBullish and engulf and bodyPerc >= minEngulfBodyPerc

// =====================================================
// FUNCIÓN ENVOLVENTE BAJISTA
// =====================================================

bearEngulfingFunc() =>
    prevBullish = close[1] > open[1]

    currentBearish = close < open

    body = math.abs(close - open)

    candleRange = high - low

    bodyPerc = candleRange > 0 ? body / candleRange * 100 : 0.0

    engulf = open >= close[1] and close <= open[1]

    prevBullish and currentBearish and engulf and bodyPerc >= minEngulfBodyPerc

// =====================================================
// FUNCIÓN MARTILLO
// =====================================================

hammerFunc() =>
    body = math.abs(close - open)

    upperWick = high - math.max(open, close)

    lowerWick = math.min(open, close) - low

    candleRange = high - low

    bodyPerc = candleRange > 0 ? body / candleRange * 100 : 0.0

    validBody = body > 0

    validBody and lowerWick >= body * wickBodyRatio and upperWick <= body * oppositeWickMaxRatio and bodyPerc <= maxBodyRangePerc

// =====================================================
// FUNCIÓN SHOOTING STAR
// =====================================================

shootingStarFunc() =>
    body = math.abs(close - open)

    upperWick = high - math.max(open, close)

    lowerWick = math.min(open, close) - low

    candleRange = high - low

    bodyPerc = candleRange > 0 ? body / candleRange * 100 : 0.0

    validBody = body > 0

    validBody and upperWick >= body * wickBodyRatio and lowerWick <= body * oppositeWickMaxRatio and bodyPerc <= maxBodyRangePerc

bool nearResistanceZone = false

float bestResistanceDistance = 1000000.0

    for i = 0 to array.size(resistancePrices) - 1

        resistancePrice = array.get(resistancePrices, i)

        resistanceUpper = resistancePrice * (1 + zoneWidthPerc / 100)

        resistanceLower = resistancePrice * (1 - zoneWidthPerc / 100)

        extraMargin = useExtraProximityFilter ? resistancePrice * (patternProximityPerc / 100) : 0.0

        resistanceUpperExtended = resistanceUpper + extraMargin

        resistanceLowerExtended = resistanceLower - extraMargin

        nearResistance = patternHigh >= resistanceLowerExtended and patternLow <= resistanceUpperExtended

        if not nearResistance
            array.set(resistanceUsed, i, false)

        if nearResistance

            currentDistance = math.abs(patternHigh - resistancePrice) / resistancePrice * 100

            if currentDistance < bestResistanceDistance

                bestResistanceDistance := currentDistance

                resistanceMatchIndex := i

                nearResistanceZone := true

// =====================================================
// BUSCAR ZONA DE SOPORTE CERCANA
// =====================================================

bool nearSupportZone = false

int supportMatchIndex = na

        supportUpper = supportPrice * (1 + zoneWidthPerc / 100)

        supportLower = supportPrice * (1 - zoneWidthPerc / 100)

        extraMargin = useExtraProximityFilter ? supportPrice * (patternProximityPerc / 100) : 0.0

        if not nearSupport

        if nearSupport

            if currentDistance < bestSupportDistance

                bestSupportDistance := currentDistance

                supportMatchIndex := i

                nearSupportZone := true
